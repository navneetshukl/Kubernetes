1. Load image in minikube node
minikube image load notification-api:latest

2. Describe pod status
kubectl describe pod notification-api-pod

3. Logs inside pod
kubectl logs notification-api-pod

4. Delete image from minikube
minikube image rm notification-api:latest

5. Testing curl command

# Create notification
curl -X POST http://notification-api-service:8000/notifications \
  -H "Content-Type: application/json" \
  -d '{"user_id": "101", "message": "Your order has shipped"}'

  curl -X POST http://notification-api-service:8000/notifications \
  -H "Content-Type: application/json" \
  -d '{"user_id": "102", "message": "Your order has shipped"}'

  curl -X POST http://notification-api-service:8000/notifications \
  -H "Content-Type: application/json" \
  -d '{"user_id": "103", "message": "Your order has shipped"}'

# Get all notifications
curl -X GET http://localhost:8080/notifications

# Test validation error

6. Enter inside the pod
kubectl exec -it <pod-name> -- /bin/sh

7. Logs for specific container inside pod
kubectl logs <pod-name> -c <container-name>

8. Temporary curl pod for debugging
kubectl run curl-test --image=curlimages/curl --rm -it -- sh
